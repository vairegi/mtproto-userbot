# ---------------------------------------------------------------------------
# Cloudflare bypass for hf_scraper.py (added 2026-07 after Render 403s).
#
# hentaifox.com sits behind Cloudflare with an active JS challenge on every
# request. Datacenter IPs (Render, Fly.io, HF Spaces) get an immediate HTTP
# 403 with `cf-mitigated: challenge` when hit by plain httpx / requests /
# even curl_cffi with TLS impersonation — we proved this during debugging.
#
# cloudscraper actually runs the JS challenge and stores the resulting
# cf_clearance cookie so subsequent requests pass through. Version 1.2.71
# is the current stable release on PyPI (the "3.0.0" tag on GitHub was never
# published there — pip install would fail). Verified working against
# hentaifox.com from a datacenter IP.
# ---------------------------------------------------------------------------
cloudscraper==1.2.71

# ---------------------------------------------------------------------------
# MongoDB — replaces the old local SQLite queue.db file.
#
# Version note (verified by actually installing these together):
#   motor 3.6.0 requires pymongo>=4.9,<4.10 — so pymongo MUST be 4.9.x here.
#   Using pymongo 4.10+ with motor 3.6.0 makes pip fail with
#   "ResolutionImpossible", which would break the build on Hugging Face.
#
# The `[srv]` extra is what enables "mongodb+srv://" connection strings, which
# is the format MongoDB Atlas gives you. From pymongo 4.9 onward, the required
# dnspython package became a normal dependency, so the extra is a harmless
# no-op (pip may print "does not provide the extra 'srv'" — safe to ignore).
# dnspython is pinned explicitly below so mongodb+srv:// always resolves.
# ---------------------------------------------------------------------------
pymongo[srv]==4.9.2
dnspython==2.7.0
motor==3.6.0
